cmake_minimum_required(VERSION 3.24)
project(PathwayDictationEngine LANGUAGES C CXX)

if(NOT APPLE AND NOT WIN32)
    message(FATAL_ERROR "Dictation engines currently support macOS and Windows.")
endif()

set(PATHWAY_ENGINE "speech" CACHE STRING "speech or cleanup; build separately because ggml revisions differ")
set(PATHWAY_ENGINE_OUTPUT "${CMAKE_CURRENT_SOURCE_DIR}/../build/engines" CACHE PATH "Packaged worker directory")
option(PATHWAY_ENGINE_GPU "Compile the platform GPU backend (Metal on Mac, Vulkan on Windows)" ${APPLE})

set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
if(APPLE)
    set(GGML_METAL ${PATHWAY_ENGINE_GPU} CACHE BOOL "" FORCE)
    set(GGML_ACCELERATE ON CACHE BOOL "" FORCE)
    set(GGML_BLAS ON CACHE BOOL "" FORCE)
    set(GGML_BLAS_VENDOR Apple CACHE STRING "" FORCE)
else()
    set(GGML_VULKAN ${PATHWAY_ENGINE_GPU} CACHE BOOL "" FORCE)
    # CPU fallback runs on baseline Windows x64; optimized release builds may override.
    set(GGML_AVX OFF CACHE BOOL "")
    set(GGML_AVX2 OFF CACHE BOOL "")
    set(GGML_FMA OFF CACHE BOOL "")
    set(GGML_F16C OFF CACHE BOOL "")
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

include(FetchContent)
# Archives have no .git directory. Do not accidentally report Pathway's parent
# checkout as the upstream ggml revision in the compiled library.
set(GIT_EXE "" CACHE FILEPATH "" FORCE)
set(GIT_EXECUTABLE "" CACHE FILEPATH "" FORCE)
if(PATHWAY_ENGINE STREQUAL "speech")
    set(GGML_BUILD_COMMIT "371b5a7561823ab2bb32142d2751e35e7534727b")
    set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
    set(WHISPER_CURL OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(whisper
        URL https://codeload.github.com/ggml-org/whisper.cpp/tar.gz/371b5a7561823ab2bb32142d2751e35e7534727b
        URL_HASH SHA256=89051d8fca516a3ad1f5c2f8f9d2fccb089afbaec338fca3f8731999babc6f81
        DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_MakeAvailable(whisper)
    set(worker pathway-speech-engine)
    add_executable(${worker} speech/worker.cpp)
    target_include_directories(${worker} SYSTEM PRIVATE "${whisper_SOURCE_DIR}/examples")
    target_link_libraries(${worker} PRIVATE whisper)
    set(dependency_source "${whisper_SOURCE_DIR}")
    set(dependency_name whisper)
elseif(PATHWAY_ENGINE STREQUAL "cleanup")
    set(GGML_BUILD_COMMIT "b95502ba9aa0eb73a2f4fc8878d7fbe6a847a0b9")
    foreach(flag COMMON TESTS TOOLS EXAMPLES SERVER APP UI)
        set(LLAMA_BUILD_${flag} OFF CACHE BOOL "" FORCE)
    endforeach()
    set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(llama
        URL https://codeload.github.com/ggml-org/llama.cpp/tar.gz/b95502ba9aa0eb73a2f4fc8878d7fbe6a847a0b9
        URL_HASH SHA256=c2ec2a837346b7ecb2a0ff4e2ac343667067b02dd795f296573d50bdf11cc37f
        DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_MakeAvailable(llama)
    set(worker pathway-cleanup-engine)
    add_executable(${worker} cleanup/worker.cpp)
    target_include_directories(${worker} SYSTEM PRIVATE "${llama_SOURCE_DIR}/vendor/nlohmann")
    target_link_libraries(${worker} PRIVATE llama)
    set(dependency_source "${llama_SOURCE_DIR}")
    set(dependency_name llama)
else()
    message(FATAL_ERROR "PATHWAY_ENGINE must be speech or cleanup")
endif()

target_compile_features(${worker} PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(${worker} PRIVATE /W4 /utf-8)
    target_compile_definitions(${worker} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
else()
    target_compile_options(${worker} PRIVATE -Wall -Wextra -Wpedantic)
endif()
set_target_properties(${worker} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PATHWAY_ENGINE_OUTPUT}")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(${worker} PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${PATHWAY_ENGINE_OUTPUT}")
endforeach()

# Keep notices next to the binaries for desktop packaging. Models are downloaded only at runtime.
file(MAKE_DIRECTORY "${PATHWAY_ENGINE_OUTPUT}/notices")
file(COPY "${CMAKE_CURRENT_SOURCE_DIR}/notices/" DESTINATION "${PATHWAY_ENGINE_OUTPUT}/notices")
configure_file("${dependency_source}/LICENSE" "${PATHWAY_ENGINE_OUTPUT}/notices/${dependency_name}-LICENSE.txt" COPYONLY)
if(EXISTS "${dependency_source}/ggml/LICENSE")
    file(COPY "${dependency_source}/ggml/LICENSE" DESTINATION "${PATHWAY_ENGINE_OUTPUT}/notices/${dependency_name}-ggml")
endif()
if(EXISTS "${dependency_source}/licenses")
    file(COPY "${dependency_source}/licenses/" DESTINATION "${PATHWAY_ENGINE_OUTPUT}/notices/${dependency_name}-dependencies")
endif()
